<!-- CellEditingPage.xaml. Text columns get a built-in TextBox editor; the Department column
     supplies its own editor through CellEditingTemplate. -->
<DataTemplate x:Key="DepartmentTextTemplate" x:DataType="models:Person">
    <TextBlock Text="{x:Bind Department, Mode=OneWay}" Margin="12,0,8,0"
               VerticalAlignment="Center" TextTrimming="CharacterEllipsis"/>

<!-- Classic {Binding} with UpdateSourceTrigger=Explicit and no x:DataType: the control commits by
     calling UpdateSource on the editor's binding expressions, which it can find only on a
     {Binding}. Esc therefore discards the choice without touching the model. -->
<DataTemplate x:Key="DepartmentEditingTemplate">
    <ComboBox SelectedItem="{Binding Department, Mode=TwoWay, UpdateSourceTrigger=Explicit}"
              AutomationProperties.Name="Department"
              HorizontalAlignment="Stretch" VerticalAlignment="Center" Margin="4,0">
        <x:String>Design</x:String>
        <x:String>Engineering</x:String>
        <!-- ...the other PersonData.Departments values... -->
    </ComboBox>
</DataTemplate>
