<!-- Display cell of the Department column. -->
<DataTemplate x:Key="DepartmentTextTemplate" x:DataType="models:Person">
    <TextBlock Text="{x:Bind Department, Mode=OneWay}" Margin="12,0,8,0"

</DataTemplate>

<!-- Editor of the Department column, shown only while the cell is in edit mode. It uses a
     classic {Binding} with UpdateSourceTrigger=Explicit, and no x:DataType: the control
     commits by calling UpdateSource on the editor's binding expressions, which it can find
     only on a {Binding}. Esc therefore discards the choice without touching the model.
     The items are the PersonData.Departments values. -->
<DataTemplate x:Key="DepartmentEditingTemplate">
    <ComboBox SelectedItem="{Binding Department, Mode=TwoWay, UpdateSourceTrigger=Explicit}"
              AutomationProperties.Name="Department"
              HorizontalAlignment="Stretch" VerticalAlignment="Center" Margin="4,0">
        <x:String>Design</x:String>
        <x:String>Engineering</x:String>
        <x:String>Finance</x:String>
        <x:String>HR</x:String>
        <x:String>Marketing</x:String>
        <x:String>Operations</x:String>
        <x:String>Product</x:String>
        <x:String>Sales</x:String>
    </ComboBox>
</DataTemplate>
