InitializeComponent();

public ObservableCollection<Person> People { get; } = PersonData.Take(RowCount);

public CollectionViewSource Source { get; }

private void OnFilterChanged(object sender, TextChangedEventArgs e)
{
    var text = ((TextBox)sender).Text.Trim();
    if (text.Length == 0)
    {
        Source.Filter = null;
    }
    else
    {
        // The view filters before the rows are nested: a match whose parent is filtered out is a root.
        Source.Filter = new CollectionViewFilter(item => ((Person)item).FullName.Contains(text, StringComparison.CurrentCultureIgnoreCase));
    }
}

private void OnSortChanged(object sender, SelectionChangedEventArgs e)
{
    // The view sorts; nesting keeps that order within each sibling set.
    Source.SortDescriptions.Clear();
    switch (((ComboBox)sender).SelectedIndex)
    {
        case 1:
            Source.SortDescriptions.Add(new SortDescription(nameof(Person.FullName), ListSortDirection.Ascending));
            break;
        case 2:
            Source.SortDescriptions.Add(new SortDescription(nameof(Person.FullName), ListSortDirection.Descending));
            break;
        case 3:
            Source.SortDescriptions.Add(new SortDescription(nameof(Person.Role), ListSortDirection.Ascending));
            break;
    }
}

// TableView.Columns is the live column collection: moving an entry moves the column. The first
// visible column hosts the level indent and the expander chevron.
private void SetFirstColumn(bool photoFirst)
{
    var columns = PeopleTable.Columns;
    var from = columns.IndexOf(PhotoColumn);
    if (from < 0 || (from == 0) == photoFirst)
    {
        return;
    }

    // Move only Photo; the other columns keep their order.
    columns.RemoveAt(from);
    columns.Insert(photoFirst ? 0 : columns.IndexOf(NameColumn) + 1, PhotoColumn);
}

// From Controls\ShapingOptions.xaml.cs
// ItemKeyPath names each row's key and ParentKeyPath its parent's key. Changing either replaces
// the relation and resets expansion, so set them once.
private static void NestRows(TableView table, ParentRelation relation)
{
    table.ItemKeyPath = relation.KeyPath;
    table.ParentKeyPath = relation.ParentKeyPath;
}

// From Controls\ShapingOptions.xaml.cs
// Under GroupBy + ParentBy, ExpandAllGroups affects only the group headers and ExpandAllRows only
// the tree rows, so Grouped hierarchy calls both.
private static void ExpandEvery(TableView table, bool grouped, bool hierarchical)
{
    if (grouped)
    {
        table.ExpandAllGroups();
    }

    if (hierarchical)
    {
        table.ExpandAllRows();
    }
}

private static void CollapseEvery(TableView table, bool grouped, bool hierarchical)
{
    if (grouped)
    {
        table.CollapseAllGroups();
    }

    if (hierarchical)
    {
        table.CollapseAllRows();
    }
}

// From Helpers\ParentRelation.cs
// The two paths the table takes: a row's unique key, and its parent's key (null for a root).
// Person rows: EmployeeId under ManagerId (the org chart).
public static ParentRelation Manager { get; } = new(
    "Manager",
    nameof(Person.EmployeeId),
    nameof(Person.ManagerId),
    row => (row as Person)?.EmployeeId,
    row => (row as Person)?.ManagerId);

// Person rows: EmployeeId under MentorId, a second tree over the same rows.
public static ParentRelation Mentor { get; } = new(
    "Mentor",
    nameof(Person.EmployeeId),
    nameof(Person.MentorId),
    row => (row as Person)?.EmployeeId,
    row => (row as Person)?.MentorId);

// From Models\Person.cs
// The Name column binds FullName. Its setter lets page actions rename a person: the first word
// becomes FirstName and the rest LastName. Each part raises PropertyChanged, and so does FullName.
public string FullName
{
    get => $"{_firstName} {_lastName}".Trim();
    set
    {
        var name = (value ?? string.Empty).Trim();
        var split = name.IndexOfAny(new[] { ' ', '\t' });
        FirstName = split < 0 ? name : name[..split];
        LastName = split < 0 ? string.Empty : name[(split + 1)..].Trim();
    }
}
