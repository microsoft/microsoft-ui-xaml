<!-- TableView.EmptyTemplate renders (centered over the row area) whenever
     ItemsSource is null or resolves to zero rows. Keep the content to text +
     shapes — a FontIcon inside EmptyTemplate crashes at startup. -->

<Page.Resources>
    <DataTemplate x:Key="EmptyStateTemplate">
        <StackPanel HorizontalAlignment="Center" VerticalAlignment="Center" Spacing="12">
            <Grid Width="64" Height="64" HorizontalAlignment="Center">
                <Ellipse Fill="{ThemeResource SubtleFillColorSecondaryBrush}"
                         Stroke="{ThemeResource CardStrokeColorDefaultBrush}" StrokeThickness="1"/>
                <Rectangle Width="26" Height="20" RadiusX="3" RadiusY="3"
                           Stroke="{ThemeResource TextFillColorTertiaryBrush}" StrokeThickness="2"/>
            </Grid>
            <TextBlock Text="No rows to display" HorizontalAlignment="Center"
                       Style="{ThemeResource SubtitleTextBlockStyle}"/>
            <TextBlock Text="The table's ItemsSource is null or empty. Load the sample data."
                       TextAlignment="Center" TextWrapping="Wrap"
                       Foreground="{ThemeResource TextFillColorSecondaryBrush}"/>
        </StackPanel>
    </DataTemplate>
</Page.Resources>

<tv:TableView x:Name="DemoTable"
               SelectionMode="Single"
               EmptyTemplate="{StaticResource EmptyStateTemplate}">
    <tv:TableViewTextColumn Header="Name"       Binding="{Binding FullName}"/>
    <tv:TableViewTextColumn Header="Department" Binding="{Binding Department}"/>
    <tv:TableViewTextColumn Header="Role"       Binding="{Binding Role}"/>
    <tv:TableViewTextColumn Header="Email"      Binding="{Binding Email}"/>
</tv:TableView>

<!-- Three data sources, because there are two different ways of being empty:
     a null ItemsSource, and a live shaped source that holds no items. -->
<muxc:RadioButtons x:Name="DataSourceRadioButtons"
                   SelectedIndex="0"
                   SelectionChanged="OnDataSourceSelectionChanged"
                   AutomationProperties.Name="Data source">
    <RadioButton Content="Populated"                  Tag="Populated"/>
    <RadioButton Content="Empty (null ItemsSource)"   Tag="Null"/>
    <RadioButton Content="Empty (source with 0 rows)" Tag="ZeroRows"/>
</muxc:RadioButtons>
