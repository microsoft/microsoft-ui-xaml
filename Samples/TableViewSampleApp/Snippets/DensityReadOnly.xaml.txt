<!-- Density and IsReadOnly are table-level properties. This page demonstrates
     ordinary template TextBoxes, not framework edit transactions. The separate
     Cell editing page uses built-in TableViewTextColumn editing and Escape cancel. -->

<Page.Resources>
    <local:ReadOnlyState x:Key="ReadOnlyState"/>

    <DataTemplate x:Key="FirstNameCell">
        <TextBox Text="{Binding FirstName, Mode=TwoWay}"
                 AutomationProperties.Name="First name"
                 IsReadOnly="{Binding IsReadOnly, Source={StaticResource ReadOnlyState}}"
                 BorderThickness="0" Background="Transparent"/>
    </DataTemplate>
</Page.Resources>

<tabular:TableView x:Name="DemoTable" SelectionMode="Single">
    <tabular:TableViewTextColumn     Header="Name"       Binding="{Binding FullName}"/>
    <tabular:TableViewTemplateColumn Header="First name" SortMemberPath="FirstName"
                                 CellTemplate="{StaticResource FirstNameCell}"/>
    <!-- Last name / Role / Email columns follow the same editable pattern -->
</tabular:TableView>
