<Page xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
      xmlns:tv="using:Microsoft.UI.Xaml.Controls.Tabular"
      xmlns:models="using:TableViewSampleApp.Models"
      xmlns:local="using:TableViewSampleApp.Pages">

  <Page.Resources>
    <!-- Converters return SHARED, cached brushes; they never allocate in Convert,
         which would churn a brush per cell under virtualization. -->
    <local:SortStandingTintConverter x:Key="SortStandingTintConverter"/>
    <local:SortStandingDotConverter  x:Key="SortStandingDotConverter"/>
    <local:SortStandingTextConverter x:Key="SortStandingTextConverter"/>

    <DataTemplate x:Key="StandingChipTemplate" x:DataType="models:LeagueTeam">
      <Border Background="{x:Bind Points, Mode=OneWay, Converter={StaticResource SortStandingTintConverter}}"
              CornerRadius="10" Padding="8,2" HorizontalAlignment="Left">
        <StackPanel Orientation="Horizontal" Spacing="6">
          <!-- Decorative: hidden from the automation tree. -->
          <Ellipse Width="8" Height="8"
                   Fill="{x:Bind Points, Mode=OneWay, Converter={StaticResource SortStandingDotConverter}}"
                   AutomationProperties.AccessibilityView="Raw"/>
          <TextBlock Text="{x:Bind Points, Mode=OneWay, Converter={StaticResource SortStandingTextConverter}}"
                     Foreground="{ThemeResource TextFillColorPrimaryBrush}"/>
        </StackPanel>
      </Border>
    </DataTemplate>
  </Page.Resources>

  <tv:TableView
      x:Name="TeamsTable"
      SelectionMode="Single"
      Sorted="OnTableSorted">

    <!-- SortMemberPath identifies the property to sort by. When omitted,
         TableViewTextColumn falls back to its Binding.Path. -->
    <tv:TableViewTextColumn Header="Group"
                              SortMemberPath="Group"
                              Binding="{Binding Group}"/>
    <tv:TableViewTextColumn Header="Team"
                              SortMemberPath="Team"
                              Binding="{Binding Team}"/>

    <!-- CanSort=False hides the SortIndicator and makes the header
         non-interactive. Programmatic SortByColumn can still target it. -->
    <tv:TableViewTextColumn Header="P"
                              CanSort="False"
                              Binding="{Binding Played}"/>

    <tv:TableViewTextColumn Header="W"
                              SortMemberPath="Wins"
                              Binding="{Binding Wins}"/>
    <tv:TableViewTextColumn Header="D"
                              SortMemberPath="Draws"
                              Binding="{Binding Draws}"/>
    <tv:TableViewTextColumn Header="L"
                              SortMemberPath="Losses"
                              Binding="{Binding Losses}"/>
    <tv:TableViewTextColumn Header="GF"
                              SortMemberPath="GoalsFor"
                              Binding="{Binding GoalsFor}"/>
    <tv:TableViewTextColumn Header="GA"
                              SortMemberPath="GoalsAgainst"
                              Binding="{Binding GoalsAgainst}"/>
    <tv:TableViewTextColumn Header="GD"
                              SortMemberPath="GoalDifference"
                              Binding="{Binding GoalDifference}"/>
    <tv:TableViewTextColumn Header="Pts"
                              SortMemberPath="Points"
                              Binding="{Binding Points}"/>

    <!-- A templated cell does NOT opt out of sorting: SortMemberPath names the
         property behind the template, so the chip column sorts like any other. -->
    <tv:TableViewTemplateColumn Header="Standing"
                              CanSort="True"
                              SortMemberPath="Points"
                              CellTemplate="{StaticResource StandingChipTemplate}"/>
  </tv:TableView>

  <!-- Shaping mode. Hierarchy is not available in this release, so those two
       options ship disabled with a tooltip rather than hidden or as enabled
       no-ops; re-enable them when the control ships hierarchy support. -->
  <ComboBox x:Name="ShapingModeSelector" Header="Shaping mode"
            SelectedIndex="0" SelectionChanged="OnShapingModeChanged">
    <ComboBoxItem Content="Flat"              Tag="flat"/>
    <ComboBoxItem Content="Grouped"           Tag="grouped"/>
    <ComboBoxItem Content="Hierarchy"         Tag="hierarchy"        IsEnabled="False"
                  ToolTipService.ToolTip="Hierarchical rows are not available in this release."/>
    <ComboBoxItem Content="Grouped hierarchy" Tag="groupedhierarchy" IsEnabled="False"
                  ToolTipService.ToolTip="Hierarchical rows are not available in this release."/>
  </ComboBox>

</Page>

  Header click   → ToggleSortDirection(col). Opens at Ascending, then cycles
                   Ascending -> Descending and replaces any prior sort
                   (single-column).
  SortByColumn   → Sets the sort to one column + direction in a single call.
  ClearSort      → Clears the sort; Sorted fires with no sorted columns.
  Sort + group   → Sorting a GROUPED source reorders rows WITHIN each group;
                   the grouping is not disturbed.
-->
