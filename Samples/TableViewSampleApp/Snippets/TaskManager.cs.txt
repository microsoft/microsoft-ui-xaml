// Live shaping: a row moves, leaves or changes group as soon as a CPU value, a status or a
// category changes, with no Refresh(). Sorting and grouping follow whatever the header click
// or the Shaping options choose; the search observes the properties it reads that can change.
Source = new CollectionViewSource
{
    Source = Processes,  // created once; TableView shows its View
    IsLiveSortingRequested = true,
    IsLiveGroupingRequested = true,
    IsLiveFilteringRequested = true,
};
Source.LiveFilteringProperties.Add(nameof(ProcessItem.Category));
Source.LiveFilteringProperties.Add(nameof(ProcessItem.StatusText));
InitializeComponent();

ProcessTable.HeadersVisibility = TableViewHeadersVisibility.Column;
ProcessTable.GridLinesVisibility = TableViewGridLinesVisibility.Horizontal;
ProcessTable.Density = TableViewDensity.Compact;
// Metrics open descending, as in Task Manager: the busiest process comes first.
foreach (var column in new[] { CpuColumn, MemoryColumn, DiskColumn, NetworkColumn })
{
    column.SortCycle = TableViewSortCycle.DescendingAscending;
}

RestoreColumnOrder(AppSettings.LoadColumnOrder(ColumnOrderSetting));
ProcessTable.ColumnReordered += OnProcessTableColumnReordered;

public ObservableCollection<ProcessItem> Processes { get; } = ProcessData.All();

public CollectionViewSource Source { get; }

private Task _columnOrderSave = Task.CompletedTask;

// The order is read on the UI thread; the file is written off it, one save after another.
private void QueueColumnOrderSave(IReadOnlyList<string> order)
{
    _columnOrderSave = _columnOrderSave.ContinueWith(previous =>
    {
        if (previous.Exception is { } error)
        {
            System.Diagnostics.Debug.WriteLine($"Saving the column order failed: {error.InnerException?.Message}");
        }
        AppSettings.SaveColumnOrder(ColumnOrderSetting, order);
    }, TaskScheduler.Default);
}

// Keyed by header text, which this sample does not localize; a real app keys on something stable.
private void OnProcessTableColumnReordered(TableView sender, TableViewColumnReorderedEventArgs args)
{
    QueueColumnOrderSave(sender.Columns.Select(c => c.Header as string).ToList());
}

private void RestoreColumnOrder(IReadOnlyList<string> order)
{
    // Known columns only, once each: a stale or repeated saved name must not shift the rest.
    // Frozen columns keep their leading prefix; the saved order applies to the columns after it.
    int frozen = ProcessTable.Columns.TakeWhile(c => c.FrozenEdge == TableViewFrozenEdge.Leading).Count();
    var known = order.Distinct()
        .Select(header => ProcessTable.Columns.FirstOrDefault(c => c.Header as string == header))
        .Where(column => column is not null && column.FrozenEdge != TableViewFrozenEdge.Leading)
        .ToList();
    for (int i = 0; i < known.Count; i++)
    {
        int target = frozen + i;
        int current = ProcessTable.Columns.IndexOf(known[i]);
        if (current != target)
        {
            ProcessTable.Columns.RemoveAt(current);
            ProcessTable.Columns.Insert(target, known[i]);
        }
    }
}

private void OnSearchBoxTextChanged(AutoSuggestBox sender, AutoSuggestBoxTextChangedEventArgs args)
{
    if (args.Reason != AutoSuggestionBoxTextChangeReason.UserInput)
    {
        return;
    }

    var query = sender.Text?.Trim();
    _searchText = string.IsNullOrEmpty(query) ? null : query;
    ApplySearch();
}

// Filter is evaluated when the view reshapes and, with live filtering, when Category or StatusText
// changes on a row (LiveFilteringProperties).
private void ApplySearch()
{
    if (_searchText is not { } text)
    {
        ((ICollectionViewShaping)Source.View).Filter = null;
        return;
    }

    ((ICollectionViewShaping)Source.View).Filter = new CollectionViewFilter(item => item is ProcessItem process && MatchesSearch(process, text));
}

private static bool MatchesSearch(ProcessItem process, string query) =>
    process.Name.Contains(query, StringComparison.CurrentCultureIgnoreCase)
    || process.Category.Contains(query, StringComparison.CurrentCultureIgnoreCase)
    || process.StatusText.Contains(query, StringComparison.CurrentCultureIgnoreCase);
