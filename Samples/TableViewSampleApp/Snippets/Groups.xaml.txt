<Page
    xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
    xmlns:models="using:TableViewSampleApp.Models"
    xmlns:tv="using:Microsoft.UI.Xaml.Controls.Tabular">

  <Page.Resources>
    <!-- Left-aligned StackPanel, not a stretched Grid: the group-header ContentPresenter
         sizes header content to its desired width, so a Grid with a `*` first column
         collapsed the count and the chip to zero width and they never rendered.
         Binds ItemCount (Int32) so the count is always populated. -->
    <DataTemplate x:Key="CustomGroupHeaderTemplate"
                  x:DataType="tv:TableViewGroupInfo">
      <StackPanel Orientation="Horizontal" Spacing="10" Padding="0,6">
        <TextBlock Text="{x:Bind KeyText, Mode=OneWay}"
                   Style="{ThemeResource BodyStrongTextBlockStyle}"/>
        <TextBlock Text="{x:Bind ItemCount, Mode=OneWay,
                          Converter={StaticResource GroupCountTextConverter}}"
                   Style="{ThemeResource CaptionTextBlockStyle}"/>
      </StackPanel>
    </DataTemplate>

    <!-- In-cell editors are REAL controls: default IsTabStop, no IsHitTestVisible="False",
         Mode=TwoWay, and Person raises PropertyChanged for each bound property so a
         recycled row re-reads the committed value. Each carries a per-row accessible name. -->
    <DataTemplate x:Key="DepartmentCellTemplate" x:DataType="models:Person">
      <ComboBox ItemsSource="{StaticResource DepartmentChoices}"
                SelectedItem="{x:Bind Department, Mode=TwoWay}"
                AutomationProperties.Name="{Binding FullName,
                    Converter={StaticResource RowAutomationNameConverter},
                    ConverterParameter='Department for {0}'}"/>
    </DataTemplate>

    <DataTemplate x:Key="ActiveCellTemplate" x:DataType="models:Person">
      <CheckBox IsChecked="{x:Bind IsActive, Mode=TwoWay}"
                AutomationProperties.Name="{Binding FullName,
                    Converter={StaticResource RowAutomationNameConverter},
                    ConverterParameter='Active for {0}'}"/>
    </DataTemplate>
  </Page.Resources>

  <tv:TableView x:Name="PeopleTableControl" SelectionMode="Single">
    <tv:TableView.Columns>
      <tv:TableViewTextColumn Header="First name" Binding="{Binding FirstName}"
                               SortMemberPath="FirstName"/>
      <tv:TableViewTemplateColumn Header="Department"
                                   HeaderToolTip="The grouped-on value when the group key is Department."
                                   CellTemplate="{StaticResource DepartmentCellTemplate}"
                                   CanSort="True" SortMemberPath="Department"/>
      <tv:TableViewTemplateColumn Header="Active"
                                   CellTemplate="{StaticResource ActiveCellTemplate}"
                                   CanSort="True" SortMemberPath="IsActive"/>
    </tv:TableView.Columns>
  </tv:TableView>

</Page>
