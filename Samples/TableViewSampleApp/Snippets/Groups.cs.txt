InitializeComponent();
ApplyGroupHeaderTemplate();

public ObservableCollection<Person> People { get; }

public TableViewSource Source { get; }

private void ApplyGroupHeaderTemplate()
{
    PeopleTable.GroupHeaderTemplate = (string)((ComboBoxItem)HeaderTemplateSelector.SelectedItem).Tag == "custom"
        ? (DataTemplate)Resources["CustomGroupHeaderTemplate"]
        : null;
}

// TableView.Columns is the live column collection: moving an entry moves the column. The first
// visible column hosts the group indent and the expander chevron.
private void SetFirstColumn(TableViewColumn first)
{
    var columns = PeopleTable.Columns;
    var from = columns.IndexOf(first);
    if (from > 0)
    {
        columns.RemoveAt(from);
        columns.Insert(0, first);
    }
}

// From Controls\ShapingOptions.xaml.cs
// The key selector receives the ROW and the identity selector the KEY. GroupBy takes a delegate,
// so call it again after a row's grouped-on value changes.
private static void GroupRows(TableViewSource source, Func<object?, object> groupKeyOf) =>
    source.GroupBy(row => groupKeyOf(row), SampleShaping.GroupIdentity);

// From Helpers\SampleShaping.cs
// Bucket label for a null or blank group key.
public const string NoneKey = "(none)";

// From Helpers\SampleShaping.cs
// GroupBy identity selector: it receives the group key (never the row) and must not return an
// empty string, because TableViewSource fails fast on an empty identity.
public static string GroupIdentity(object? groupKey) =>
    groupKey?.ToString() is { Length: > 0 } s ? s : NoneKey;
