private void ApplyGroupHeaderTemplate()
{
    PeopleTable.GroupHeaderTemplate = (string)((ComboBoxItem)HeaderTemplateSelector.SelectedItem).Tag == "custom"
        ? (DataTemplate)Resources["CustomGroupHeaderTemplate"]
        : null;
}

// TableView.Columns is the live column collection: moving an entry moves the column. The first
// visible column hosts the group indent and the expander chevron.
private void SetFirstColumn(bool photoFirst)
{
    var columns = PeopleTable.Columns;
    var from = columns.IndexOf(PhotoColumn);
    if (from < 0 || (from == 0) == photoFirst)
    {
        return;
    }

    // Move only Photo; the other columns keep their order.
    columns.RemoveAt(from);
    columns.Insert(photoFirst ? 0 : columns.IndexOf(NameColumn) + 1, PhotoColumn);
}

// From Controls\ShapingOptions.xaml.cs
// A PropertyGroupDescription names the property to group by. Grouping reads the value when the
// view refreshes, so call Refresh() after a row's grouped-on value changes.
private static void GroupRows(CollectionViewSource source, string groupPath)
{
    var description = new PropertyGroupDescription(groupPath);
    if (source.GroupDescriptions.Count == 1)
    {
        source.GroupDescriptions[0] = description;
    }
    else
    {
        source.GroupDescriptions.Clear();
        source.GroupDescriptions.Add(description);
    }
}

// Turning grouping on or off gives the CollectionViewSource a new View, so show the current one.
private static void ShowView(TableView table, CollectionViewSource source)
{
    if (!ReferenceEquals(table.ItemsSource, source.View))
    {
        table.ItemsSource = source.View;
    }
}

// From Controls\ShapingOptions.xaml.cs
// Under GroupBy + ParentBy, ExpandAllGroups affects only the group headers and ExpandAllRows only
// the tree rows, so Grouped hierarchy calls both.
private static void ExpandEvery(TableView table, bool grouped, bool hierarchical)
{
    if (grouped)
    {
        table.ExpandAllGroups();
    }

    if (hierarchical)
    {
        table.ExpandAllRows();
    }
}

private static void CollapseEvery(TableView table, bool grouped, bool hierarchical)
{
    if (grouped)
    {
        table.CollapseAllGroups();
    }

    if (hierarchical)
    {
        table.CollapseAllRows();
    }
}

// From Helpers\SampleShaping.cs
// The Person property a group-key Tag groups by: IsActive groups by its Active/Inactive label.
public static string GroupPathOf(string key) => key switch
{
    nameof(Person.Office) => nameof(Person.Office),
    nameof(Person.Role) => nameof(Person.Role),
    nameof(Person.IsActive) => nameof(Person.ActiveStatus),
    _ => nameof(Person.Department),
};

// From Models\Person.cs
// The Name column binds FullName. Its setter lets page actions rename a person: the first word
// becomes FirstName and the rest LastName. Each part raises PropertyChanged, and so does FullName.
public string FullName
{
    get => $"{_firstName} {_lastName}".Trim();
    set
    {
        var name = (value ?? string.Empty).Trim();
        var split = name.IndexOfAny(new[] { ' ', '\t' });
        FirstName = split < 0 ? name : name[..split];
        LastName = split < 0 ? string.Empty : name[(split + 1)..].Trim();
    }
}
