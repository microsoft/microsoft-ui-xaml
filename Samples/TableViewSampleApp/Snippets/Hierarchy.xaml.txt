    VerticalAlignment="Stretch"
    MinHeight="320">
    <tv:TableView.Resources>
        <x:Double x:Key="TableViewRowIndentSize">24</x:Double>
        <!-- One menu for every row. Items tagged Leaf (individual contributors) or
             Manager (rows with reports) show only for that row kind; untagged
             items show for both. OnRowMenuOpening picks the subset from the
             row the menu opens on, so a row that gains its first report gets
             the manager items the next time its menu opens. Menu items inherit
             the row's Employee as DataContext, so their state can bind to it:
             Promote binds the Employee's PromoteCommand (MVVM, disabled unless
             CanPromote), and exactly one of Assign / Remove mentor shows,
             depending on HasMentor. -->
        <MenuFlyout x:Key="RowMenu" Opening="OnRowMenuOpening">
            <MenuFlyoutItem Text="Open profile" Click="OnRowMenuItemClick"/>
            <MenuFlyoutItem Text="Copy row" Click="OnCopyRowClick"/>
            <MenuFlyoutItem Text="Email team" Tag="Manager" Click="OnRowMenuItemClick"/>
            <MenuFlyoutItem Text="Promote" Tag="Leaf" Command="{Binding PromoteCommand}"/>
            <MenuFlyoutSeparator Tag="Leaf"/>
            <MenuFlyoutItem Text="Assign mentor" Tag="Leaf" Click="OnAssignMentorClick"
                            Visibility="{Binding HasMentor, Converter={StaticResource InvertedBoolToVisibilityConverter}}"/>
            <MenuFlyoutItem Text="Remove mentor" Tag="Leaf" Click="OnRemoveMentorClick"
                            Visibility="{Binding HasMentor, Converter={StaticResource BoolToVisibilityConverter}}"/>
            <MenuFlyoutSeparator Tag="Leaf"/>
            <MenuFlyoutItem Text="Remove" Tag="Leaf" Click="OnRowMenuItemClick"/>
        </MenuFlyout>
        <!-- A group header's DataContext is its TableViewGroupInfo. -->
        <MenuFlyout x:Key="DeptGroupMenu">
            <MenuFlyoutItem Text="{Binding KeyText}" IsEnabled="False"/>
            <MenuFlyoutItem Text="Email department" Click="OnRowMenuItemClick"/>
        </MenuFlyout>
        <Style TargetType="tv:TableViewRow">
            <Setter Property="ContextFlyout" Value="{StaticResource RowMenu}"/>
        </Style>
        <!-- Department headers (Group on): the style owns the band chrome;
             GroupHeaderTemplate (set in code) still owns the content. -->
        <Style TargetType="tv:TableViewGroupHeader">
            <Setter Property="ContextFlyout" Value="{StaticResource DeptGroupMenu}"/>
            <Setter Property="Background" Value="{ThemeResource SystemFillColorAttentionBackgroundBrush}"/>
            <Setter Property="Foreground" Value="{ThemeResource AccentTextFillColorPrimaryBrush}"/>
            <Setter Property="Padding" Value="8,10,16,10"/>
        </Style>
    </tv:TableView.Resources>
</tv:TableView>
