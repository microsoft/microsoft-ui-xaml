<!-- The tree is the CollectionViewSource's: a flat process list, nested by parent process id. TableView has no
     hierarchy of its own. The sort orders every sibling list; live sorting moves a process among its siblings. -->
<CollectionViewSource x:Name="Processes" Filter="OnProcessFilter" IsLiveSortingRequested="True">
    <CollectionViewSource.HierarchyDescription>
        <PropertyHierarchyDescription ItemKeyPath="Pid" ParentKeyPath="ParentPid"/>
    </CollectionViewSource.HierarchyDescription>
    <CollectionViewSource.SortDescriptions>
        <SortDescription PropertyName="CpuPercent" Direction="Descending"/>
    </CollectionViewSource.SortDescriptions>
</CollectionViewSource>
<DataTemplate x:Key="TreeProcessNameTemplate" x:DataType="models:ProcessItem">

        <TextBlock Text="{x:Bind Name}" TextTrimming="CharacterEllipsis" VerticalAlignment="Center"/>
    </StackPanel>
</DataTemplate>
<DataTemplate x:Key="TreeCpuTemplate" x:DataType="models:ProcessItem">
    <Border Background="{x:Bind CpuPercent, Mode=OneWay, Converter={StaticResource ProcessHeatBrushConverter}, ConverterParameter=Cpu}" Padding="8,4">
        <TextBlock Text="{x:Bind CpuDisplay, Mode=OneWay}" HorizontalAlignment="Right" VerticalAlignment="Center"/>
    </Border>
</DataTemplate>
<DataTemplate x:Key="TreeMemoryTemplate" x:DataType="models:ProcessItem">

    <tv:TableViewTemplateColumn Header="CPU" Width="100" CellTemplate="{StaticResource TreeCpuTemplate}" SortMemberPath="CpuPercent"/>
