<!-- HeadersVisibility gates the COLUMN header strip only. Group header bands are a separate
     surface and stay when it is None. IsReadOnly gates the built-in text editors; the Due and
     Resolved template editors stay editable. -->

              ItemsSource="{x:Bind Source}"
              SelectionMode="Single"
              IsReadOnly="True">

    <tv:TableViewTemplateColumn Header="Assignee" Width="{StaticResource ColWidthNumber}" CellTemplate="{StaticResource AssigneeTemplate}" SortMemberPath="Assignee" CellToolTipBinding="{Binding Assignee}"/>
    <tv:TableViewTextColumn Header="Summary" Width="*" MinWidth="{StaticResource ColMinWidthStar}" Binding="{Binding Summary}" CellToolTipBinding="{Binding Summary}"/>
    <tv:TableViewTemplateColumn Header="Priority" Width="{StaticResource ColWidthNumber}" CellTemplate="{StaticResource PriorityChipTemplate}" SortMemberPath="Priority"/>
    <tv:TableViewTemplateColumn x:Name="DueColumn" Header="Due" Width="{StaticResource ColWidthCalendarDate}" CellTemplate="{StaticResource DueDateTemplate}" SortMemberPath="Due"/>
