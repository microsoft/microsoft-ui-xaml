                                 HeaderToolTip="Urgency: Sev 1 is the most urgent. Editable."/>

                                 HeaderToolTip="Checked once the ticket is closed. Editable."/>
</tv:TableView>

<!-- Cell editors are REAL editors: TwoWay against a property that raises
     PropertyChanged (so the value survives row recycling), IsTabStop left at
     its default so they are keyboard reachable, and a per-row accessible name
     so twenty of them do not announce identically. -->
<DataTemplate x:Key="PriorityCell" x:DataType="local:SupportTicket">
    <ComboBox ItemsSource="{x:Bind PriorityOptions}"
              SelectedItem="{x:Bind Priority, Mode=TwoWay}"
              AutomationProperties.Name="{x:Bind PriorityAutomationName, Mode=OneWay}"/>
</DataTemplate>

<DataTemplate x:Key="ResolvedCell" x:DataType="local:SupportTicket">
    <CheckBox IsChecked="{x:Bind IsResolved, Mode=TwoWay}"
              AutomationProperties.Name="{x:Bind ResolvedAutomationName, Mode=OneWay}"/>
</DataTemplate>
