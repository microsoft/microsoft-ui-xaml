// There is nothing to switch on. TableView virtualizes by default, so the
// only consumer-facing lesson here is how NOT to defeat it.

// DON'T: per-item Add into an already-bound collection. Each Add raises a
// CollectionChanged notification on the UI thread, so 50,000 rows means
// 50,000 layout-invalidating notifications and a frozen window.
foreach (var row in rows) { People.Add(row); }     // <- stalls for seconds

// DO: build the set off the bind path, then assign once.
var rows = new List<NumberedPerson>(count);
for (int i = 0; i < count; i++) { rows.Add(/* ... */); }

People = new ObservableCollection<NumberedPerson>(rows);
PeopleTable.ItemsSource = People;                  // one reset notification

// grid the widths re-derive from whatever is on screen and shift as you
// scroll. Prefer Pixel or Star widths for large sets.
