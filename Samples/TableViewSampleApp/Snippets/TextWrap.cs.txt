ApplyTextState();

// Single owner of the shared TextWrapState. Both the wrap toggle and the max-lines
// picker funnel through here, so neither overwrites the other's contribution, and every
// value lands on the state object the cell DataTemplate binds to -- which is what makes
// the setting survive row recycling.
private void ApplyTextState()
{
    var wrapOn = WrapToggle?.IsOn ?? true;
    var state = GetTextWrapState();

    state.TextWrapping = wrapOn ? TextWrapping.Wrap : TextWrapping.NoWrap;

    // TextBlock.MaxLines only clamps text that actually wraps, so it is deliberately not
    // applied in NoWrap mode -- and the picker is disabled to say so.
    state.MaxLines = wrapOn ? _maxLines : 0;
    state.TextTrimming = !wrapOn || _maxLines > 0
        ? TextTrimming.CharacterEllipsis
        : TextTrimming.None;

    MaxLinesComboBox.IsEnabled = wrapOn;
}

private void OnWrapToggled(object sender, RoutedEventArgs e)
{
    ApplyTextState();
}

private void OnMaxLinesSelectionChanged(object sender, SelectionChangedEventArgs e)
{
    var item = (ComboBoxItem)MaxLinesComboBox.SelectedItem;
    _maxLines = int.Parse((string)item.Tag, CultureInfo.InvariantCulture);
    ApplyTextState();
}

private void OnColumnWidthChanged(object sender, RangeBaseValueChangedEventArgs e)
{
    _columnWidth = e.NewValue;
    BioColumn.Width = new GridLength(_columnWidth);
}
