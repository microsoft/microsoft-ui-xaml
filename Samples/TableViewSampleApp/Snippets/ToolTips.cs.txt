// Both tooltip surfaces are opt-in and both are per column.

// HeaderToolTip takes any object. A header is not bound against a row, so there
// is nothing to defer and no binding involved. A string is reported as help text
// alongside the sort state; non-string content is mouse-only.
SalaryColumn.HeaderToolTip = "Salary";

var card = new StackPanel { Spacing = 4 };
card.Children.Add(new TextBlock { Text = "Bio", FontWeight = FontWeights.SemiBold });
card.Children.Add(new TextBlock { Text = "Free-form summary", Opacity = 0.75 });
BioColumn.HeaderToolTip = card;

// CellToolTipBinding is a Binding evaluated against each row's data item, which
// is why it is a CLR property rather than a DP: XAML has to hand the Binding
// over unevaluated instead of resolving it once.
BioColumn.CellToolTipBinding = new Binding { Path = new PropertyPath(nameof(Person.Bio)) };
DepartmentColumn.CellToolTipBinding = new Binding { Path = new PropertyPath(nameof(Person.Role)) };

// No Path: the binding produces the row item, and a converter turns it into
// rich content. Build a FRESH element per evaluation — the returned UIElement is
// parented by the cell's ToolTip, and one element cannot have two parents.
NameColumn.CellToolTipBinding = new Binding { Converter = new RowCardConverter() };

// Clearing the binding retracts the tooltip; there is no separate remove call.
NameColumn.CellToolTipBinding = null;

// The Name, Department and Salary columns are sortable (they do not opt out with
// CanSort="False"), so the header gesture has to actually do something.
SampleShape.EnableDefaults(PeopleTable, _people);

// Shaping is independent of the tooltips: the columns are untouched by a
// reshape, so every HeaderToolTip and CellToolTipBinding survives it, and group
// header bands are not cells so they never acquire a cell tooltip.
_source = TableViewSource.From(_people);
PeopleTable.ItemsSource = _source;

private void ApplyShaping()
{
    if (_mode != "grouped")
    {
        _source.ClearGroupBy();
    }
    else
    {
        var key = _groupKey;
        //   TableViewKeySelector(Object item)      -> receives the ROW ITEM
        //   TableViewIdentitySelector(Object item) -> receives the GROUP KEY
        _source.GroupBy(
            item => (object)GroupValue(item, key),
            groupKey => groupKey?.ToString() ?? "(none)");
    }

    // case "hierarchy":
    // case "groupedhierarchy":
    //     Hierarchical rows are not available in this release: neither
    //     TableViewSource.idl nor TableView.idl exposes a hierarchy verb (the
    //     only shaping verbs are Filter / GroupBy / Sort and their Clear*
    //     counterparts). When the control ships hierarchy support, apply it to
    //     this same source here, alongside the GroupBy stage above.

    _appliedMode = _mode;   // only after the call returns
}
