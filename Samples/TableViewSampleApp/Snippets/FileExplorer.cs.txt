// FileSystemEntry.cs: folders-first sort keys. The "0 "/"1 " prefix keeps folders together, and
// the fixed-width digits make the size compare as a number, not as display text.
public string NameSortKey => SortPrefix + Name;
public string SizeSortKey => SortPrefix + Size.ToString("D19", CultureInfo.InvariantCulture);
private string SortPrefix => IsFolder ? "0 " : "1 ";

// DateTimeOffset, not DateTime: it projects to Windows.Foundation.DateTime, which the sort
// compares as a point in time. A System.DateTime has no WinRT projection and would sort as text.
public DateTimeOffset DateModified { get; set; }

// FileExplorerPage.xaml.cs. One source over one collection; a folder's entries replace its items.
public ObservableCollection<FileSystemEntry> Entries { get; } = new();

public FileExplorerPage()
{
    _source = TableViewSource.From(Entries);
    InitializeComponent();
    // handledEventsToo: the table consumes Enter for its own row navigation.
    FileTable.AddHandler(KeyDownEvent, new KeyEventHandler(OnTableKeyDown), handledEventsToo: true);
    Loaded += OnPageLoaded;  // opens FileSystemBrowser.InitialDirectory (AppContext.BaseDirectory)
}

// The folder is read on a background thread; only the result is applied on the UI thread.
private async Task NavigateAsync(string path, bool addToHistory, string verb)
{
    var version = ++_navigationVersion;
    var listing = await Task.Run(() => FileSystemBrowser.Read(path));

    // A newer navigation started while this one was reading; drop the stale result.
    if (version != _navigationVersion)
    {
        return;
    }

    if (listing.Error is not null)
    {
        AddressBar.Text = _currentDir;
        SetLastAction(string.Format(CultureInfo.CurrentCulture, "Cannot open {0}: {1}", listing.FullPath, listing.Error));
        return;
    }

    // ...push history, then refill the one collection.
    _currentDir = listing.FullPath;
    var focusState = TableFocusState();
    Entries.Clear();
    foreach (var entry in listing.Entries)
    {
        Entries.Add(entry);
    }

    // Refilling recycles every row; move focus and selection to the new folder's first row.
    if (focusState != FocusState.Unfocused)
    {
        FocusFirstRowAfterLayout(focusState);
    }
}

// Enter opens the selected folder; Backspace goes up.
private void OnTableKeyDown(object sender, KeyRoutedEventArgs e)
{
    if (e.Key == VirtualKey.Enter)
    {
        e.Handled = true;
        OpenSelectedFolder();
    }
    else if (e.Key == VirtualKey.Back)
    {
        e.Handled = true;
        GoUp();
    }
}

// Grouped by date: a sort declared BEFORE GroupBy orders the groups, so sorting newest first
// makes the buckets read Today, Yesterday, ... The key selector receives the ROW; the identity
// selector receives the KEY.
if (key == nameof(FileSystemEntry.DateModified))
{
    FileTable.SortByColumn(DateModifiedColumn, SortDirection.Descending);
}

_source.GroupBy(item => KeyOf(item as FileSystemEntry, key), SampleShaping.GroupIdentity);
